   144→function CategorySidebar({
   145→  selected,
   146→  onChange,
   147→  onNavigate,
   148→}: {
   149→  selected: ShopCategory[] | null;
   150→  onChange: (cats: ShopCategory[] | null) => void;
   151→  onNavigate?: () => void;
   152→}) {
   153→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   154→
   155→  const toggle = (slug: string) => {
   156→    setOpenNodes((prev) => {
   157→      const next = new Set(prev);
   158→      if (next.has(slug)) next.delete(slug);
   159→      else next.add(slug);
   160→      return next;
   161→    });
   162→  };
   163→
   164→  function renderNode(node: CategoryNode, depth: number) {
   165→    const hasChildren = node.children && node.children.length > 0;
   166→    const isOpen = openNodes.has(node.slug);
   167→    const isLeaf = !!node.categories;
   168→
   169→    return (
   170→      <div key={node.slug}>
   171→        <button
   172→          onClick={() => {
   173→            if (isLeaf) {
   174→              onChange(node.categories!);
   175→              onNavigate?.();
   176→            } else if (hasChildren) toggle(node.slug);
   177→          }}
   178→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   179→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   180→              ? "text-black font-medium"
   181→              : "text-black"
   182→          }`}
   183→          style={{ paddingLeft: `${depth * 14}px` }}
   184→        >
   185→          <span>{node.label}</span>
   186→          {hasChildren && (
   187→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   188→          )}
   189→        </button>
   190→        {hasChildren && isOpen && (
   191→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   192→        )}
   193→      </div>
   194→    );
   195→  }
   196→
   197→  return (
   198→    <div className="w-full">
   199→      <div className="flex items-center justify-between mb-4">
   200→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   201→        {selected && (
   202→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   203→        )}
   204→      </div>
   205→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   206→    </div>
   207→  );
   208→}

   214→const SIZE_LABELS: Record<string, string> = {

   224→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   225→  let activeSizeType: SizeType = "none";
   226→  if (categories) {
   227→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   228→    if (types.size === 1) activeSizeType = [...types][0];
   229→    else activeSizeType = "clothing";
   230→  }
   231→
   232→  const options = SIZE_OPTIONS[activeSizeType];
   233→  const label = SIZE_LABELS[activeSizeType];
   234→  if (!options || options.length === 0 || !label) return null;
   235→
   236→  return (
   237→    <div className="mt-6">
   238→      <p className="text-[10px] uppercase tracking-[0.2em] text-black mb-3">{label}</p>
   239→      <div className="flex flex-wrap gap-2">
   240→        {options.map((s) => (
   241→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   242→        ))}
   243→      </div>
   244→    </div>
   245→  );

   573→export function ProductDetailPage({ slug }: { slug?: string }) {
   574→  const router = useRouter();

   707→   DETAIL ROUTER — checks both data sources
   708→   ═══════════════════════════════════════════════════ */
   709→
   710→export function ShopDetailRouter({ slug }: { slug: string }) {
   711→  // Check product-catalog (POD/merch) first
   712→  const catalogProduct = getProductById(slug);
   713→  if (catalogProduct) {
   714→    return <PortalProductDetailPage productId={slug} />;
   715→  }
   716→
   717→  // Fall back to shop catalog (albums + merch)
   718→  const isAlbum = SHOP_ALBUMS.some((a) => a.slug === slug);
   719→  const isProduct = ALL_PRODUCTS.some((p) => p.slug === slug);
   720→
   721→  if (isAlbum) return <AlbumDetailPage slug={slug} />;
   722→  if (isProduct) return <ProductDetailPage slug={slug} />;
   723→
   724→  return (
   725→    <div className="flex min-h-screen items-center justify-center bg-white">
   726→      <p className="text-sm text-black">Item not found.</p>
   727→    </div>
   728→  );
   729→}
