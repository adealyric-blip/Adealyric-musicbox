   146→function CategorySidebar({
   147→  selected,
   148→  onChange,
   149→  onNavigate,
   150→}: {
   151→  selected: ShopCategory[] | null;
   152→  onChange: (cats: ShopCategory[] | null) => void;
   153→  onNavigate?: () => void;
   154→}) {
   155→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   156→
   157→  const toggle = (slug: string) => {
   158→    setOpenNodes((prev) => {
   159→      const next = new Set(prev);
   160→      if (next.has(slug)) next.delete(slug);
   161→      else next.add(slug);
   162→      return next;
   163→    });
   164→  };
   165→
   166→  function renderNode(node: CategoryNode, depth: number) {
   167→    const hasChildren = node.children && node.children.length > 0;
   168→    const isOpen = openNodes.has(node.slug);
   169→    const isLeaf = !!node.categories;
   170→
   171→    return (
   172→      <div key={node.slug}>
   173→        <button
   174→          onClick={() => {
   175→            if (isLeaf) {
   176→              onChange(node.categories!);
   177→              onNavigate?.();
   178→            } else if (hasChildren) toggle(node.slug);
   179→          }}
   180→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   181→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   182→              ? "text-black font-medium"
   183→              : "text-black"
   184→          }`}
   185→          style={{ paddingLeft: `${depth * 14}px` }}
   186→        >
   187→          <span>{node.label}</span>
   188→          {hasChildren && (
   189→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   190→          )}
   191→        </button>
   192→        {hasChildren && isOpen && (
   193→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   194→        )}
   195→      </div>
   196→    );
   197→  }
   198→
   199→  return (
   200→    <div className="w-full">
   201→      <div className="flex items-center justify-between mb-4">
   202→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   203→        {selected && (
   204→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   205→        )}
   206→      </div>
   207→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   208→    </div>
   209→  );
   210→}

   216→const SIZE_LABELS: Record<string, string> = {
   217→  clothing: "Clothing Size",

   226→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   227→  let activeSizeType: SizeType = "none";
   228→  if (categories) {
   229→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   230→    if (types.size === 1) activeSizeType = [...types][0];
   231→    else activeSizeType = "clothing";
   232→  }
   233→
   234→  const options = SIZE_OPTIONS[activeSizeType];
   235→  const label = SIZE_LABELS[activeSizeType];
   236→  if (!options || options.length === 0 || !label) return null;
   237→
   238→  return (
   239→    <div className="mt-6">
   240→      <p className="text-[10px] uppercase tracking-[0.2em] text-black mb-3">{label}</p>
   241→      <div className="flex flex-wrap gap-2">
   242→        {options.map((s) => (
   243→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   244→        ))}
   245→      </div>
   246→    </div>
