   138→function CategorySidebar({
   139→  selected,
   140→  onChange,
   141→  onNavigate,
   142→}: {
   143→  selected: ShopCategory[] | null;
   144→  onChange: (cats: ShopCategory[] | null) => void;
   145→  onNavigate?: () => void;
   146→}) {
   147→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   148→
   149→  const toggle = (slug: string) => {
   150→    setOpenNodes((prev) => {
   151→      const next = new Set(prev);
   152→      if (next.has(slug)) next.delete(slug);
   153→      else next.add(slug);
   154→      return next;
   155→    });
   156→  };
   157→
   158→  function renderNode(node: CategoryNode, depth: number) {
   159→    const hasChildren = node.children && node.children.length > 0;
   160→    const isOpen = openNodes.has(node.slug);
   161→    const isLeaf = !!node.categories;
   162→
   163→    return (
   164→      <div key={node.slug}>
   165→        <button
   166→          onClick={() => {
   167→            if (isLeaf) {
   168→              onChange(node.categories!);
   169→              onNavigate?.();
   170→            } else if (hasChildren) toggle(node.slug);
   171→          }}
   172→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   173→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   174→              ? "text-black font-medium"
   175→              : "text-black"
   176→          }`}
   177→          style={{ paddingLeft: `${depth * 14}px` }}
   178→        >
   179→          <span>{node.label}</span>
   180→          {hasChildren && (
   181→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   182→          )}
   183→        </button>
   184→        {hasChildren && isOpen && (
   185→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   186→        )}
   187→      </div>
   188→    );
   189→  }
   190→
   191→  return (
   192→    <div className="w-full">
   193→      <div className="flex items-center justify-between mb-4">
   194→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   195→        {selected && (
   196→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   197→        )}
   198→      </div>
   199→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   200→    </div>
   201→  );
   202→}

   208→const SIZE_LABELS: Record<string, string> = {
   209→  clothing: "Clothing Size",
   210→  waist: "Waist Size",
   211→  shoe: "Shoe Size",
   212→  hat: "Hat Size",
   213→  drinkware: "Size",
   214→  "one-size": "",

   217→
   218→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   219→  let activeSizeType: SizeType = "none";
   220→  if (categories) {
   221→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   222→    if (types.size === 1) activeSizeType = [...types][0];
   223→    else activeSizeType = "clothing";
   224→  }
   225→
   226→  const options = SIZE_OPTIONS[activeSizeType];
   227→  const label = SIZE_LABELS[activeSizeType];
   228→  if (!options || options.length === 0 || !label) return null;
   229→
   230→  return (
   231→    <div className="mt-6">
   232→      <p className="text-[10px] uppercase tracking-[0.2em] text-black mb-3">{label}</p>
   233→      <div className="flex flex-wrap gap-2">
   234→        {options.map((s) => (
   235→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   236→        ))}
   237→      </div>
   238→    </div>
   239→  );
   240→}

   562→export function ProductDetailPage({ slug }: { slug?: string }) {
   563→  const router = useRouter();
   564→  const { detailSlug, setDetailSlug, addToCart, setActiveTab, setShopSubPath } = useAppStore();
   565→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   566→  const [notifyEmail, setNotifyEmail] = useState("");
   567→  const [notifySent, setNotifySent] = useState(false);
   568→  const [sidebarCats, setSidebarCats] = useState<ShopCategory[] | null>(null);
   569→
   570→  const activeSlug = slug || detailSlug;
   571→  const product = useMemo(() => ALL_PRODUCTS.find((p) => p.slug === activeSlug), [activeSlug]);
   572→  const available = product ? product.stock > 0 : false;
   573→  const enriched = product ? enrichProduct(product as ShopProduct) : null;
   574→
   575→  const related = useMemo(() => {
   576→    if (!product) return [];
   577→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   578→  }, [product]);
   579→
   580→  const openRelated = useCallback((p: EnrichedProduct) => {
   581→    router.push(`/shop/${p.slug}`);
   582→    setSelectedSize(null);
   583→    setNotifySent(false);
   584→  }, [router]);
   585→
   586→  const handleAddToCart = useCallback(() => { if (available) addToCart(); }, [available, addToCart]);
   587→  const handleNotify = useCallback(() => { if (notifyEmail) setNotifySent(true); }, [notifyEmail]);
   588→
   589→  if (!product || !enriched) {
   590→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black">Product not found.</p></div>;
   591→  }
   592→
   593→  return (

   595→      <div className="mx-auto max-w-[1400px]">

   597→        <div className="mb-8 flex items-center gap-6">
   598→          <button
   599→            onClick={() => {
   600→              const parentAlbum = SHOP_ALBUMS.find((a) => a.slug === product.albumSlug);
   601→              if (parentAlbum) { router.push(`/shop/${parentAlbum.slug}`); }
   602→              else { router.push("/shop"); }
   603→            }}
   604→            className="flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer"
   605→          >
   606→            <ArrowLeft className="h-4 w-4" />
   607→            Back
   608→          </button>
   609→          <button onClick={() => { router.push("/shop"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Shop All</button>
   610→          <button onClick={() => { router.push("/shop"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Collections</button>
   611→        </div>
   612→
   613→        <div className="grid grid-cols-1 gap-8 md:grid-cols-[180px_1fr_1fr] md:gap-0">
   614→          {/* Category Sidebar */}
   615→          <div className="hidden md:block pr-8 border-r border-black/10">
   616→            <CategorySidebar selected={sidebarCats} onChange={setSidebarCats} onNavigate={() => { setDetailSlug(null, null); setActiveTab("shop"); }} />
   617→          </div>
   618→
   619→          {/* Image */}
   620→          <div className="relative aspect-square overflow-hidden bg-[#f4f4f4]">
   621→            {product.image ? (
   622→              <img src={product.image} alt={product.name} className="h-full w-full object-cover" />

   624→              <div className="flex h-full w-full items-center justify-center"><ShoppingBag className="h-12 w-12 text-black/10" /></div>

   637→            <p className="mt-4 text-2xl font-light">{available ? `$${product.price}.00` : "Sold Out"}</p>
