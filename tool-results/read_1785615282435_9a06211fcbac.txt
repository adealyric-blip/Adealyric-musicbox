   137→function CategorySidebar({
   138→  selected,
   139→  onChange,
   140→  onNavigate,
   141→}: {
   142→  selected: ShopCategory[] | null;
   143→  onChange: (cats: ShopCategory[] | null) => void;
   144→  onNavigate?: () => void;
   145→}) {
   146→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   147→
   148→  const toggle = (slug: string) => {
   149→    setOpenNodes((prev) => {
   150→      const next = new Set(prev);
   151→      if (next.has(slug)) next.delete(slug);
   152→      else next.add(slug);
   153→      return next;
   154→    });
   155→  };
   156→
   157→  function renderNode(node: CategoryNode, depth: number) {
   158→    const hasChildren = node.children && node.children.length > 0;
   159→    const isOpen = openNodes.has(node.slug);
   160→    const isLeaf = !!node.categories;
   161→
   162→    return (
   163→      <div key={node.slug}>
   164→        <button
   165→          onClick={() => {
   166→            if (isLeaf) {
   167→              onChange(node.categories!);
   168→              onNavigate?.();
   169→            } else if (hasChildren) toggle(node.slug);
   170→          }}
   171→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   172→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   173→              ? "text-black font-medium"
   174→              : "text-black"
   175→          }`}
   176→          style={{ paddingLeft: `${depth * 14}px` }}
   177→        >
   178→          <span>{node.label}</span>
   179→          {hasChildren && (
   180→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   181→          )}
   182→        </button>
   183→        {hasChildren && isOpen && (
   184→          <div>{(node.children ?? []).map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   185→        )}
   186→      </div>
   187→    );
   188→  }
   189→
   190→  return (
   191→    <div className="w-full">
   192→      <div className="flex items-center justify-between mb-4">
   193→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   194→        {selected && (
   195→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   196→        )}
   197→      </div>
   198→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   199→    </div>
   200→  );
   201→}

   207→const SIZE_LABELS: Record<string, string> = {
   208→  clothing: "Clothing Size",
   209→  waist: "Waist Size",
   210→  shoe: "Shoe Size",
   211→  hat: "Hat Size",
   212→  drinkware: "Size",
   213→  "one-size": "",
   214→  none: "",

   216→
   217→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   218→  let activeSizeType: SizeType = "none";
   219→  if (categories) {
   220→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   221→    if (types.size === 1) activeSizeType = [...types][0];
   222→    else activeSizeType = "clothing";
   223→  }
   224→
   225→  const options = SIZE_OPTIONS[activeSizeType];
   226→  const label = SIZE_LABELS[activeSizeType];
   227→  if (!options || options.length === 0 || !label) return null;
   228→
   229→  return (
   230→    <div className="mt-6">
   231→      <p className="text-[10px] uppercase tracking-[0.2em] text-black mb-3">{label}</p>
   232→      <div className="flex flex-wrap gap-2">
   233→        {options.map((s) => (
   234→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   235→        ))}
   236→      </div>
   237→    </div>
   238→  );
   239→}

   364→export function ShopPage() {
   365→  const { setDetailSlug, addToCart, cartCount, shopSubPath, setShopSubPath } = useAppStore();
   366→  const [shopPath, setShopPath] = useState<ShopPath>((shopSubPath as ShopPath) || "albums");

   373→  }, [shopSubPath]);
   374→
   375→  const openAlbum = useCallback((album: ShopAlbum) => { setDetailSlug(album.slug, "album"); }, [setDetailSlug]);
   376→  const openProduct = useCallback((p: ShopProduct) => { setDetailSlug(p.slug, "product"); }, [setDetailSlug]);
   377→  const handleQuickAdd = useCallback(() => { addToCart(); }, [addToCart]);
   378→
   379→  return (
   380→    <section className="min-h-screen bg-white px-6 pt-36 pb-20 md:px-12 md:pt-48 md:pb-28">
   381→      <div className="mx-auto max-w-[1400px]">

   385→          <ShopNav active={shopPath} onChange={setShopPath} cartCount={cartCount} />

   558→export function ProductDetailPage() {

   560→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   561→  const [notifyEmail, setNotifyEmail] = useState("");

   563→  const [sidebarCats, setSidebarCats] = useState<ShopCategory[] | null>(null);
   564→

   571→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   572→  }, [product]);
   573→

   584→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black">Product not found.</p></div>;

   599→          >
   600→            <ArrowLeft className="h-4 w-4" />

   602→          </button>

   608→          {/* Category Sidebar */}
   609→          <div className="hidden md:block pr-8 border-r border-black/10">
   610→            <CategorySidebar selected={sidebarCats} onChange={setSidebarCats} onNavigate={() => { setDetailSlug(null, null); setActiveTab("shop"); }} />
   611→          </div>

   613→          {/* Image */}
   614→          <div className="relative aspect-square overflow-hidden bg-[#f4f4f4]">
   615→            {product.image ? (
   616→              <img src={product.image} alt={product.name} className="h-full w-full object-cover" />

   632→            {enriched.availability === "Low Stock" && <p className="mt-1 text-[11px] text-black">Low Stock — only {product.stock} left</p>}
   633→            <p className="mt-4 text-[12px] font-medium uppercase tracking-[0.2em] text-black">{product.albumTitle} Collection</p>
