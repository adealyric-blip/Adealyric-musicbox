   141→function CategorySidebar({
   142→  selected,
   143→  onChange,
   144→  onNavigate,
   145→}: {
   146→  selected: ShopCategory[] | null;
   147→  onChange: (cats: ShopCategory[] | null) => void;
   148→  onNavigate?: () => void;
   149→}) {
   150→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   151→
   152→  const toggle = (slug: string) => {
   153→    setOpenNodes((prev) => {
   154→      const next = new Set(prev);
   155→      if (next.has(slug)) next.delete(slug);
   156→      else next.add(slug);
   157→      return next;
   158→    });
   159→  };
   160→
   161→  function renderNode(node: CategoryNode, depth: number) {
   162→    const hasChildren = node.children && node.children.length > 0;
   163→    const isOpen = openNodes.has(node.slug);
   164→    const isLeaf = !!node.categories;
   165→
   166→    return (
   167→      <div key={node.slug}>
   168→        <button
   169→          onClick={() => {
   170→            if (isLeaf) {
   171→              onChange(node.categories!);
   172→              onNavigate?.();
   173→            } else if (hasChildren) toggle(node.slug);
   174→          }}
   175→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   176→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   177→              ? "text-black font-medium"
   178→              : "text-black"
   179→          }`}
   180→          style={{ paddingLeft: `${depth * 14}px` }}
   181→        >
   182→          <span>{node.label}</span>
   183→          {hasChildren && (
   184→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   185→          )}
   186→        </button>
   187→        {hasChildren && isOpen && (
   188→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   189→        )}
   190→      </div>
   191→    );
   192→  }
   193→
   194→  return (
   195→    <div className="w-full">
   196→      <div className="flex items-center justify-between mb-4">
   197→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   198→        {selected && (
   199→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   200→        )}
   201→      </div>
   202→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   203→    </div>
   204→  );
   205→}

   211→const SIZE_LABELS: Record<string, string> = {
   212→  clothing: "Clothing Size",
   213→  waist: "Waist Size",
   214→  shoe: "Shoe Size",
   215→  hat: "Hat Size",
   216→  drinkware: "Size",
   217→  "one-size": "",
   218→  none: "",

   220→
   221→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   222→  let activeSizeType: SizeType = "none";
   223→  if (categories) {
   224→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   225→    if (types.size === 1) activeSizeType = [...types][0];
   226→    else activeSizeType = "clothing";
   227→  }
   228→
   229→  const options = SIZE_OPTIONS[activeSizeType];
   230→  const label = SIZE_LABELS[activeSizeType];
   231→  if (!options || options.length === 0 || !label) return null;
   232→
   233→  return (
   234→    <div className="mt-6">
   235→      <p className="text-[10px] uppercase tracking-[0.2em] text-black mb-3">{label}</p>
   236→      <div className="flex flex-wrap gap-2">
   237→        {options.map((s) => (
   238→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   239→        ))}
   240→      </div>
   241→    </div>
   242→  );
   243→}

   565→export function ProductDetailPage({ slug }: { slug?: string }) {
   566→  const router = useRouter();
   567→  const { detailSlug, setDetailSlug, addToCart, setActiveTab, setShopSubPath } = useAppStore();

   571→  const [sidebarCats, setSidebarCats] = useState<ShopCategory[] | null>(null);
   572→
   573→  const activeSlug = slug || detailSlug;
   574→  const product = useMemo(() => ALL_PRODUCTS.find((p) => p.slug === activeSlug), [activeSlug]);
   575→  const available = product ? product.stock > 0 : false;
   576→  const enriched = product ? enrichProduct(product as ShopProduct) : null;

   579→    if (!product) return [];
   580→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   581→  }, [product]);
   582→
   583→  const openRelated = useCallback((p: EnrichedProduct) => {

   585→    setSelectedSize(null);
   586→    setNotifySent(false);

   588→

   592→  if (!product || !enriched) {
   593→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black">Product not found.</p></div>;

   602→            onClick={() => {

   604→              if (parentAlbum) { router.push(`/shop/${parentAlbum.slug}`); }
   605→              else { router.push("/shop"); }
   606→            }}
   607→            className="flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer"
   608→          >
   609→            <ArrowLeft className="h-4 w-4" />
   610→            Back
   611→          </button>
